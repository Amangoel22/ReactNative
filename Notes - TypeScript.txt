1. Unlike js ensures that a varible's type is defined.

5. An object can be defined in ts by using 'interface':

    interface UserInterface{
        id: number;
        name: string;
        age: number;
        greet(message:string): void(if no return statement else type)   //also define input's type
    }
    const User:UserInterface{
        id: 2,
        name: "ABC",
        age: 20,
        greet(message){
            console.log(message)
        }
    }

7. Intersection:

    interface UserIdentity{
        id: number,
        name: string,
    }
    interface BusinessPartner{
        name: string,
        creditScore: number,
    }

    type Employee = BusinessPartner & UserIdentity  //intersection: using properties of both objects

    const signContract = (employee: Employee): void => {  //void means no return
        console.log("Contract signed by:" + employee.name)
    }

9. Generics: <T>

    class StorageContainer<T>{  //while declaring the class, we made it generic, so that if anyone wants to use it, they can use and create class data type as they want

        private contents: T[]

        constructor (){  //no arrow as it is a class method
            this.contents = [];
        }

        addItem(item: T): void{

        }
        getItem(idx: number) : T | undefined{
            return this.contents[idx];
        }
    }

    //using the above defined class
    const userNames = new StorageContainer<string>(); //we are using the class as string
    userNames.add("ABC");
